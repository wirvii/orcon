catalog_source  # SPEC-003 catalog sources of the instance

connector_revocation_override  # SPEC-003 instance-level override of a revocation
code_library  # SPEC-003 code libraries shared by the instance
session  # SPEC-004 login sessions are resolved before the organization is known
login_throttle  # SPEC-004 R-AUTH-02 failed sign-in attempts per keyed identifier, before the organization is known

instance_setup  # SPEC-015 first-run state of the instance
